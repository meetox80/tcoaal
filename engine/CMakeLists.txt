cmake_minimum_required(VERSION 3.16)
project(tcoaal CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB_RECURSE ENGINE_SRC CONFIGURE_DEPENDS src/core/*.cpp src/game/*.cpp src/main.cpp)

set(BUILD_INFO_DIR ${CMAKE_BINARY_DIR}/gen)
set(BUILD_INFO_ARGS -DSRC=${CMAKE_SOURCE_DIR}/.. -DOUT=${BUILD_INFO_DIR}/build_info.h -P ${CMAKE_SOURCE_DIR}/cmake/build_info.cmake)
execute_process(COMMAND ${CMAKE_COMMAND} ${BUILD_INFO_ARGS})
add_custom_target(build_info ALL COMMAND ${CMAKE_COMMAND} ${BUILD_INFO_ARGS} BYPRODUCTS ${BUILD_INFO_DIR}/build_info.h)

if(PSP)
  set(TCOAAL_TARGET psp)
elseif(VITASDK)
  set(TCOAAL_TARGET vita)
elseif(NINTENDO_3DS)
  set(TCOAAL_TARGET 3ds)
elseif(NINTENDO_WII)
  set(TCOAAL_TARGET wii)
elseif(WIIU)
  set(TCOAAL_TARGET wiiu)
elseif(PS2)
  set(TCOAAL_TARGET ps2)
elseif(PS3)
  set(TCOAAL_TARGET ps3)
elseif(XENON)
  set(TCOAAL_TARGET xenon)
elseif(COMMAND psn00bsdk_add_executable)
  set(TCOAAL_TARGET ps1)
else()
  set(TCOAAL_TARGET psp CACHE STRING "Console whose screen and data the desktop build reproduces (psp, vita, 3ds, wii, wiiu, ps3, ps2 or xenon)")
  if(NOT TCOAAL_TARGET MATCHES "^(psp|vita|3ds|wii|wiiu|ps3|ps2|xenon)$")
    message(FATAL_ERROR "the desktop build reproduces psp, vita, 3ds, wii, wiiu, ps3, ps2 or xenon, not ${TCOAAL_TARGET}")
  endif()
endif()
set(PROFILE_DIR ${CMAKE_SOURCE_DIR}/src/platform/${TCOAAL_TARGET})
if(NOT TCOAAL_TARGET STREQUAL "3ds")
  list(FILTER ENGINE_SRC EXCLUDE REGEX "/game/panel\\.cpp$")
endif()

if(PSP)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/psp/*.cpp src/platform/common/*.cpp)
  file(GLOB TREMOR_SRC CONFIGURE_DEPENDS third_party/tremor/*.c)
  add_library(tremor OBJECT ${TREMOR_SRC})
  target_compile_options(tremor PRIVATE -O2 -G0 -fno-strict-aliasing -w -Wno-incompatible-pointer-types)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} $<TARGET_OBJECTS:tremor>)
  target_compile_options(tcoaal PRIVATE -O2 -G0 -fno-strict-aliasing -Wall -Wno-unused-function)
  target_compile_definitions(tcoaal PRIVATE USE_TREMOR)
  target_link_libraries(tcoaal PRIVATE ogg z pspgu pspge pspdisplay pspctrl pspaudio psppower psprtc psputility pspvfpu pspdebug m)
  create_pbp_file(TARGET tcoaal TITLE "The Coffin of Andy and Leyley" VERSION "01.00" BUILD_PRX)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR} third_party)
elseif(VITASDK)
  include("${VITASDK}/share/vita.cmake" REQUIRED)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/vita/*.cpp src/platform/common/*.cpp)
  file(GLOB VITA_GXP CONFIGURE_DEPENDS src/platform/vita/shaders/*.gxp)
  set(VITA_SHADERS_SRC ${BUILD_INFO_DIR}/vita_shaders.cpp)
  string(REPLACE ";" "|" VITA_GXP_ARG "${VITA_GXP}")
  add_custom_command(OUTPUT ${VITA_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${VITA_GXP_ARG}" -DOUT=${VITA_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${VITA_GXP} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} ${VITA_SHADERS_SRC})
  target_compile_options(tcoaal PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function)
  target_link_libraries(tcoaal PRIVATE vorbisfile vorbis ogg z m SceGxm_stub SceDisplay_stub SceCtrl_stub SceAudio_stub SceAppUtil_stub SceAppMgr_stub
    SceSysmodule_stub ScePower_stub SceLibKernel_stub SceKernelThreadMgr_stub SceSysmem_stub SceIofilemgr_stub SceProcessmgr_stub)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  vita_create_self(eboot.bin tcoaal UNSAFE)
  add_custom_command(OUTPUT param.sfo
    COMMAND vita-mksfoex -s TITLE_ID=TCOAAL001 -s APP_VER=01.00 -s CATEGORY=gd -d ATTRIBUTE2=12 "The Coffin of Andy and Leyley" param.sfo VERBATIM)
  add_custom_target(param_sfo ALL DEPENDS param.sfo)
elseif(NINTENDO_3DS)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/3ds/*.cpp src/platform/common/*.cpp)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC})
  target_compile_options(tcoaal PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function)
  target_compile_definitions(tcoaal PRIVATE USE_TREMOR)
  ctr_add_shader_library(sprite ${CMAKE_CURRENT_SOURCE_DIR}/src/platform/3ds/sprite.v.pica)
  dkp_add_embedded_binary_library(sprite_shader sprite)
  target_link_libraries(tcoaal PRIVATE sprite_shader vorbisidec ogg z citro3d ctru m)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${DEVKITPRO}/portlibs/3ds/include ${BUILD_INFO_DIR} ${PROFILE_DIR})
  ctr_generate_smdh(tcoaal.smdh NAME "The Coffin of Andy and Leyley" DESCRIPTION "Unofficial 3DS port" AUTHOR "kotwicha")
  ctr_create_3dsx(tcoaal SMDH tcoaal.smdh)
  set(CTRTOOLS "" CACHE PATH "folder with makerom and bannertool for the CIA template")
  if(CTRTOOLS)
    set(CTR_DIR ${CMAKE_CURRENT_SOURCE_DIR}/ctr)
    string(ASCII 26 18 16 BANNER_PIXEL)
    string(REPEAT "${BANNER_PIXEL}" 32768 BANNER_PIXELS)
    file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/banner.ppm CONTENT "P6\n256 128\n255\n${BANNER_PIXELS}")
    add_custom_command(OUTPUT tcoaal.bnr
      COMMAND ${CTRTOOLS}/bannertool makebanner -i ${CMAKE_CURRENT_BINARY_DIR}/banner.ppm -a ${CTR_DIR}/silence.wav -o tcoaal.bnr
      DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/banner.ppm ${CTR_DIR}/silence.wav VERBATIM)
    add_custom_command(OUTPUT romfs/tcoaal.pak
      COMMAND ${CMAKE_COMMAND} -E make_directory romfs
      COMMAND ${CMAKE_COMMAND} -E echo_append TPAK > romfs/tcoaal.pak
      VERBATIM)
    add_custom_command(OUTPUT tcoaal.cia
      COMMAND ${CTRTOOLS}/makerom -f cia -o tcoaal.cia -target t -exefslogo -rsf ${CTR_DIR}/app.rsf -elf $<TARGET_FILE:tcoaal>
              -icon tcoaal.smdh -banner tcoaal.bnr -DAPP_ROMFS=romfs
      DEPENDS tcoaal ${CTR_DIR}/app.rsf tcoaal.smdh tcoaal.bnr romfs/tcoaal.pak VERBATIM)
    add_custom_target(tcoaal_cia ALL DEPENDS tcoaal.cia)
  endif()
elseif(NINTENDO_WII)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/wii/*.cpp src/platform/common/*.cpp)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC})
  target_compile_options(tcoaal PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function)
  target_link_libraries(tcoaal PRIVATE vorbisfile vorbis ogg z wiiuse bte asnd fat ogc m)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${DEVKITPRO}/portlibs/ppc/include ${PROFILE_DIR})
  ogc_create_dol(tcoaal)
elseif(WIIU)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/wiiu/*.cpp src/platform/common/*.cpp)
  file(GLOB WIIU_SHADERS CONFIGURE_DEPENDS src/platform/wiiu/shaders/*.gsh)
  set(WIIU_SHADERS_SRC ${BUILD_INFO_DIR}/wiiu_shaders.cpp)
  string(REPLACE ";" "|" WIIU_SHADERS_ARG "${WIIU_SHADERS}")
  add_custom_command(OUTPUT ${WIIU_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${WIIU_SHADERS_ARG}" -DOUT=${WIIU_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${WIIU_SHADERS} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} ${WIIU_SHADERS_SRC})
  target_compile_options(tcoaal PRIVATE -O2 -fno-strict-aliasing -fno-if-conversion -fno-if-conversion2 -Wall -Wno-unused-function)
  target_link_libraries(tcoaal PRIVATE vorbisfile vorbis ogg z m)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${DEVKITPRO}/portlibs/ppc/include ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  wut_create_rpx(tcoaal)
elseif(PS2)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/ps2/*.cpp src/platform/common/*.cpp)
  set(IRX_DIR $ENV{PS2SDK}/iop/irx)
  set(IRX_SRC)
  foreach(m iomanX fileXio libsd audsrv usbd bdm bdmfs_fatfs usbmass_bd cdfs)
    add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/irx/${m}_irx.c
      COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/irx
      COMMAND $ENV{PS2SDK}/bin/bin2c ${IRX_DIR}/${m}.irx ${CMAKE_BINARY_DIR}/irx/${m}_irx.c ${m}_irx
      DEPENDS ${IRX_DIR}/${m}.irx VERBATIM)
    list(APPEND IRX_SRC ${CMAKE_BINARY_DIR}/irx/${m}_irx.c)
  endforeach()
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} ${IRX_SRC})
  set_target_properties(tcoaal PROPERTIES SUFFIX ".elf")
  target_compile_options(tcoaal PRIVATE -O2 -G0 -fno-strict-aliasing -Wall -Wno-unused-function)
  target_link_libraries(tcoaal PRIVATE vorbisfile vorbis ogg z audsrv pad cdvd patches fileXio graph dma -ldebug kernel m)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR})
  set_source_files_properties(src/core/audio.cpp PROPERTIES COMPILE_OPTIONS -Wno-unused-variable)
elseif(PS3)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/ps3/*.cpp src/platform/common/*.cpp)
  set(SHADER_DIR ${CMAKE_SOURCE_DIR}/src/platform/ps3/shaders)
  set(SHADER_GEN ${CMAKE_BINARY_DIR}/shaders)
  set(SHADER_SRC)
  foreach(shader sprite.vp sprite.fp gray.fp)
    string(REPLACE "." "_" name ${shader})
    if(shader MATCHES "\\.vp$")
      set(kind -v)
    else()
      set(kind -f)
    endif()
    add_custom_command(OUTPUT ${SHADER_GEN}/${name}.cpp
      COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADER_GEN}
      COMMAND ${PS3DEV}/bin/cgcomp ${kind} -a ${SHADER_DIR}/${shader} ${SHADER_GEN}/${name}.bin
      COMMAND ${CMAKE_COMMAND} -DIN=${SHADER_GEN}/${name}.bin -DOUT=${SHADER_GEN}/${name}.cpp -DNAME=${name} -P ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
      DEPENDS ${SHADER_DIR}/${shader} ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
      VERBATIM)
    list(APPEND SHADER_SRC ${SHADER_GEN}/${name}.cpp)
  endforeach()
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} ${SHADER_SRC})
  set_target_properties(tcoaal PROPERTIES SUFFIX ".elf")
  target_compile_definitions(tcoaal PRIVATE PS3)
  target_compile_options(tcoaal PRIVATE -O2 -fsigned-char -fno-strict-aliasing -Wall -Wno-unused-function
    "SHELL:-include ${CMAKE_SOURCE_DIR}/src/platform/ps3/compat.h")
  target_link_libraries(tcoaal PRIVATE vorbisfile vorbis ogg z rsx gcm_sys io sysutil audio rt lv2 m)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR})
  set(CONTENT_ID UP0001-TCOAAL001_00-0000000000000000)
  add_custom_command(TARGET tcoaal POST_BUILD
    COMMAND ${CMAKE_STRIP} $<TARGET_FILE:tcoaal> -o ${CMAKE_BINARY_DIR}/tcoaal.strip.elf
    COMMAND ${PS3DEV}/bin/sprxlinker ${CMAKE_BINARY_DIR}/tcoaal.strip.elf
    COMMAND ${PS3DEV}/bin/make_self_npdrm ${CMAKE_BINARY_DIR}/tcoaal.strip.elf ${CMAKE_BINARY_DIR}/EBOOT.BIN ${CONTENT_ID}
    COMMAND ${PS3DEV}/bin/fself ${CMAKE_BINARY_DIR}/tcoaal.strip.elf ${CMAKE_BINARY_DIR}/tcoaal.self
    BYPRODUCTS ${CMAKE_BINARY_DIR}/EBOOT.BIN ${CMAKE_BINARY_DIR}/tcoaal.self
    VERBATIM)
elseif(XENON)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/xenon/*.cpp src/platform/common/*.cpp)
  file(GLOB XENON_SHADERS CONFIGURE_DEPENDS src/platform/xenon/shaders/*.vsu src/platform/xenon/shaders/*.psu)
  set(XENON_SHADERS_SRC ${BUILD_INFO_DIR}/xenon_shaders.cpp)
  string(REPLACE ";" "|" XENON_SHADERS_ARG "${XENON_SHADERS}")
  add_custom_command(OUTPUT ${XENON_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${XENON_SHADERS_ARG}" -DOUT=${XENON_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${XENON_SHADERS} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC} ${XENON_SHADERS_SRC})
  set_target_properties(tcoaal PROPERTIES SUFFIX .elf)
  target_compile_options(tcoaal PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function)
  foreach(f open read write lseek close fstat stat mkdir unlink rename ftruncate fsync opendir readdir closedir fopen fclose)
    target_link_options(tcoaal PRIVATE -Wl,--wrap=${f})
  endforeach()
  target_link_libraries(tcoaal PRIVATE -Wl,--start-group vorbisfile vorbis ogg z fat xenon m -Wl,--end-group)
  add_dependencies(tcoaal build_info)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  add_custom_command(TARGET tcoaal POST_BUILD
    COMMAND ${CMAKE_STRIP} $<TARGET_FILE:tcoaal> -o ${CMAKE_BINARY_DIR}/tcoaal.stripped.elf
    COMMAND ${CMAKE_OBJCOPY} -O elf32-powerpc --adjust-vma 0x80000000 ${CMAKE_BINARY_DIR}/tcoaal.stripped.elf ${CMAKE_BINARY_DIR}/xenon.elf
    VERBATIM)
elseif(COMMAND psn00bsdk_add_executable)
  list(FILTER ENGINE_SRC EXCLUDE REGEX "/core/(audio|fs|image|value)\\.cpp$|/game/saveio\\.cpp$")
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/ps1/*.cpp)
  psn00bsdk_add_executable(tcoaal GPREL ${ENGINE_SRC} ${PLATFORM_SRC})
  target_include_directories(tcoaal BEFORE PRIVATE src/platform/ps1/std)
  target_include_directories(tcoaal PRIVATE ${BUILD_INFO_DIR} ${PROFILE_DIR})
  add_dependencies(tcoaal build_info)
  target_compile_options(tcoaal PRIVATE -fno-strict-aliasing -Wall -Wextra -Wno-unused-parameter)
else()
  find_package(SDL2 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VORBIS REQUIRED vorbisfile)
  find_package(ZLIB REQUIRED)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/desktop/*.cpp src/platform/common/*.cpp)
  add_executable(tcoaal ${ENGINE_SRC} ${PLATFORM_SRC})
  target_include_directories(tcoaal PRIVATE third_party ${VORBIS_INCLUDE_DIRS} ${BUILD_INFO_DIR} ${PROFILE_DIR})
  add_dependencies(tcoaal build_info)
  target_link_libraries(tcoaal PRIVATE SDL2::SDL2 ${VORBIS_LIBRARIES} ZLIB::ZLIB)
  target_compile_options(tcoaal PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
  if(SANITIZE)
    target_compile_options(tcoaal PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tcoaal PRIVATE -fsanitize=address,undefined)
  endif()
endif()
